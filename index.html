function renderAlerts() { const container = document.getElementById('alertsContainer'); const list = document.getElementById('alertsList'); const badgeEl = document.getElementById('alertsFortnightBadge'); list.innerHTML = ''; const todayDay = new Date().getDate(); // Regla exacta de quincenas por día clave: // - Si el día actual está entre el 10 y el 19 -> evalúa quincena del 15 al 30 // - Si el día actual está entre el 20 y el 9 -> evalúa quincena del 1 al 14 const isLateFortnight = (todayDay >= 10 && todayDay <= 19); if (badgeEl) { badgeEl.innerText = isLateFortnight ? 'Pendientes (15-30)' : 'Pendientes (01-14)'; } let hasAlerts = false; (appData.accounts || []).forEach(acc => { const dueDay = acc.dueDay; if (!dueDay) return; const belongsToCurrentCycle = isLateFortnight ? (dueDay >= 15 && dueDay <= 31) : (dueDay >= 1 && dueDay <= 14); if (!belongsToCurrentCycle) return; const details = calculateAccountDetails(acc); const dueAmount = acc.type === 'LOAN' ? (acc.monthlyPayment || details.totalDebt) : (acc.type === 'CREDIT' ? details.cutoffDebt : details.balance); // Solo se muestra si NO está pagado y el monto a pagar es > 0 if ((acc.type === 'CREDIT' || acc.type === 'SERVICE' || acc.type === 'LOAN') && !details.isPaid && dueAmount > 0) { hasAlerts = true; let daysLeft = dueDay - todayDay; let statusText = daysLeft === 0 ? 'Vence HOY' : (daysLeft > 0 ? `Vence en ${daysLeft} días` : `Pendiente (Día ${dueDay})`); let badgeColor = daysLeft <= 2 ? 'bg-rose-500/20 text-rose-400 border-rose-500/30' : 'bg-amber-500/20 text-amber-400 border-amber-500/30'; const div = document.createElement('div'); div.className = 'flex items-center justify-between bg-slate-900/60 p-2 rounded-lg border border-slate-800'; div.innerHTML = `
${acc.name} ${acc.type === 'CREDIT' ? 'Tarjeta' : (acc.type === 'SERVICE' ? 'Servicio' : 'Crédito')}
${statusText} ${formatMoney(dueAmount, acc.currency)}
`; list.appendChild(div); } }); container.classList.toggle('hidden', !hasAlerts); } function renderDashboard() { const mainCurr = appData.mainCurrency || 'CRC'; const altCurr = mainCurr === 'CRC' ? 'USD' : 'CRC'; let totalLiquidity = 0, totalSavings = 0, totalCreditDebt = 0, totalFortnight = 0; const todayDay = new Date().getDate(); const isLateFortnight = (todayDay >= 10 && todayDay <= 19); document.getElementById('fortnightLabel').innerText = isLateFortnight ? 'A Pagar (15-30)' : 'A Pagar (01-14)'; (appData.accounts || []).forEach(acc => { const details = calculateAccountDetails(acc); if (acc.type === 'BANK' || acc.type === 'CASH') totalLiquidity += convert(details.balance, acc.currency, mainCurr); else if (acc.type === 'SAVINGS') totalSavings += convert(details.balance, acc.currency, mainCurr); else if (acc.type === 'CREDIT' || acc.type === 'LOAN' || acc.type === 'SERVICE') totalCreditDebt += convert(details.totalDebt, acc.currency, mainCurr); const dueDay = acc.dueDay; if (dueDay) { const belongs = isLateFortnight ? (dueDay >= 15 && dueDay <= 31) : (dueDay >= 1 && dueDay <= 14); const amountToPay = acc.type === 'LOAN' ? (acc.monthlyPayment || details.totalDebt) : (acc.type === 'CREDIT' ? details.cutoffDebt : details.balance); // Suma en la tarjeta "A pagar (Quincena)" únicamente si pertenece al ciclo actual, no está pagado y es > 0 if (belongs && !details.isPaid && amountToPay > 0) { totalFortnight += convert(amountToPay, acc.currency, mainCurr); } } }); const netWorth = (totalLiquidity + totalSavings) - totalCreditDebt; document.getElementById('totalNetWorth').innerText = formatMoney(netWorth, mainCurr); document.getElementById('totalNetWorthAlt').innerText = '~' + formatMoney(convert(netWorth, mainCurr, altCurr), altCurr); document.getElementById('totalLiquidity').innerText = formatMoney(totalLiquidity, mainCurr); document.getElementById('totalLiquidityAlt').innerText = '~' + formatMoney(convert(totalLiquidity, mainCurr, altCurr), altCurr); document.getElementById('totalCreditDebt').innerText = formatMoney(totalCreditDebt, mainCurr); document.getElementById('totalCreditDebtAlt').innerText = '~' + formatMoney(convert(totalCreditDebt, mainCurr, altCurr), altCurr); document.getElementById('totalSavings').innerText = formatMoney(totalSavings, mainCurr); document.getElementById('totalSavingsAlt').innerText = '~' + formatMoney(convert(totalSavings, mainCurr, altCurr), altCurr); document.getElementById('totalFortnight').innerText = formatMoney(totalFortnight, mainCurr); document.getElementById('totalFortnightAlt').innerText = '~' + formatMoney(convert(totalFortnight, mainCurr, altCurr), altCurr); renderCharts(); }
